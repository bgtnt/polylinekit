void Check(string name,double expected,double actual,bool tiny=false){checks++;double difference=Math.Abs(expected-actual);maxAbsolute=Math.Max(maxAbsolute,difference);maxRelative=Math.Max(maxRelative,difference/Math.Max(1,Math.Abs(expected)));if(!double.IsFinite(actual)||difference>1e-10*(tiny?Math.Abs(expected):Math.Max(1,Math.Abs(expected))))throw new Exception($"{name}: expected{expected:R} actual{actual:R}");}
void Compare(string name,double[] expected,double[] actual){for(int i=0;i<expected.Length;i++){
 Check(name+"/"+i,expected[i],actual[i]);
 if(name.StartsWith("single")||name.StartsWith("bridge")||(name.StartsWith("regions")&&!name.StartsWith("regions-oracle"))){double delta=Math.Abs(expected[i]-actual[i]);maxBaselineAbsolute=Math.Max(maxBaselineAbsolute,delta);maxBaselineRelative=Math.Max(maxBaselineRelative,delta/Math.Max(1,Math.Abs(expected[i])));}
}}
double[] Values(WindingAreaResult r)=>[r.NonZero,r.EvenOdd,r.AbsoluteWinding,r.Signed,r.CrossingCount];
double[] Regions(WindingOverlapResult r)=>[r.FirstArea,r.SecondArea,r.IntersectionArea,r.UnionArea,r.SymmetricDifferenceArea,r.CrossingCount];
Point2[] Rect(double x,double y,double w,double h)=>[new(x,y),new(x+w,y),new(x+w,y+h),new(x,y+h)];
Point2[] Subdivide(Point2[] p)=>p.SelectMany((a,i)=>new[]{a,new Point2((a.X+p[(i+1)%p.Length].X)/2,(a.Y+p[(i+1)%p.Length].Y)/2)}).ToArray();

}

Check("long-crossed-strips",1,WindingArea.FilledRegions(Rect(-1e16,-.5,2e16,1),Rect(-.5,-1e16,1,2e16)).IntersectionArea);

foreach(var a in duplicatePaths)foreach(var b in duplicatePaths)Compare("bridge-duplicates",baseline.Bridge(a,b),Values(WindingArea.EndpointBridged(a,b)));

Random random=new(714950);
int cases=0;
for(int trial=0;trial<2500;trial++){
 double sx=trial%3==0?1:8,sy=trial%3==0?8:1;
 Point2[] Make(int n)=>Enumerable.Range(0,n).Select(_=>new Point2((trial%2==0?random.Next(-4,5):random.NextDouble()*8-4)*sx,(trial%2==0?random.Next(-4,5):random.NextDouble()*8-4)*sy)).ToArray();
 var a=Make(random.Next(4,17));var b=Make(random.Next(4,17));if(a.Distinct().Count()<3||b.Distinct().Count()<3)continue;
 var single=WindingArea.ClosedPath(a);Compare("single"+trial,baseline.Single(a),Values(single));var slab=ContourSweep.Measure(a);Compare("slab"+trial,[slab.NonZero,slab.EvenOdd,slab.AbsoluteWinding,slab.Signed],[single.NonZero,single.EvenOdd,single.AbsoluteWinding,single.Signed]);
 Compare("bridge"+trial,baseline.Bridge(a,b),Values(WindingArea.EndpointBridged(a,b)));
 foreach(var rule in new[]{PathFillRule.NonZero,PathFillRule.EvenOdd}){
  var current=WindingArea.FilledRegions(a,b,rule);Compare("regions"+trial,baseline.Regions(a,b,rule),Regions(current));
  var reference=RegionSweep.Measure(a,b,rule==PathFillRule.NonZero);Compare("regions-oracle"+trial,[reference.First,reference.Second,reference.Intersection,reference.Union,reference.SymmetricDifference],Regions(current)[..5]);
  var reversed=WindingArea.FilledRegions(a.Reverse().ToArray(),b.Reverse().ToArray(),rule);Compare("reversed"+trial,Regions(current)[..5],Regions(reversed)[..5]);
  var divided=WindingArea.FilledRegions(Subdivide(a),Subdivide(b),rule);Compare("subdivided"+trial,Regions(current)[..5],Regions(divided)[..5]);

}
